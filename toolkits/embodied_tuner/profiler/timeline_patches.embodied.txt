#   export RLINF_TIMELINE_DIR=auto
#   export RLINF_TIMELINE_WORKER_TIMER=1
#   export RLINF_TIMELINE_ACTOR_TRAINING=1
#   export RLINF_TIMELINE_PATCH_FILE=/mnt/public/zengwen/rlinf_utils/timeline_patches.embodied.txt
#
# NOTE: Inter-component communication tracing (send_to / recv_from / chan_put /
#   chan_get across env / rollout / reward / actor) is NOT listed here. It does
#   not go through this PatchSpec file — the generic PatchSpec wrapper cannot
#   capture async wait() spans, resolve the caller component dynamically, read
#   Channel (non-Worker) context, or suppress nested re-entrant puts. It is
#   instead installed by a dedicated hook in autopatch.py and is enabled
#   automatically whenever RLINF_TIMELINE_WORKER_TIMER is on.
